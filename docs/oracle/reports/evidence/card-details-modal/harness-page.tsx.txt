import { CardDetailsProvider, useCardDetails } from '@/components/card-details/CardDetailsProvider'
import type { DeckCard } from '@/components/CardGrid'

const CARD = {
  id: 1, deck_id: 4, card_name: 'Elvish Archdruid', quantity: 1, categories: '["Ramp","Lords"]',

} as unknown as DeckCard

function Opener() {
  const api = useCardDetails()
  return <button data-testid="card-name-button" onClick={() => api?.open(1, 'Elvish Archdruid')}>Elvish Archdruid</button>
}

export default function Page() {
  return (
    <CardDetailsProvider
      deckId={4}
      cards={[CARD]}
      statusMap={new Map([[1, 'available']])}
      lifecycleMap={new Map([[1, 'planned']])}
      availableCategories={['Ramp', 'Lords', 'Draw']}
      onSaveCategories={async () => {}}
    >
      <Opener />
    </CardDetailsProvider>
  )
}
